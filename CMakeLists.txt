cmake_minimum_required(VERSION 3.16)
project(wrong_pedestrian_lights CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(Threads REQUIRED)
find_package(ZLIB REQUIRED)
find_package(BZip2 REQUIRED)
find_package(EXPAT REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SPATIALITE REQUIRED IMPORTED_TARGET spatialite sqlite3)

add_executable(wpl src/main.cpp)
target_compile_options(wpl PRIVATE -Wall -Wextra)
target_include_directories(wpl SYSTEM PRIVATE ${ZLIB_INCLUDE_DIRS} ${BZIP2_INCLUDE_DIRS} ${EXPAT_INCLUDE_DIRS})
target_link_libraries(wpl PRIVATE Threads::Threads ZLIB::ZLIB BZip2::BZip2 EXPAT::EXPAT PkgConfig::SPATIALITE)
include(GNUInstallDirs)
install(TARGETS wpl RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
